```java
wait(empty)；
wait(mutex)；
buffer[in]:= nextp；
in:=(in+1) % n；
signal(mutex)；
signal(full)；
}while(TRUE);
 }
 void consumer(){    //消费者进程

wait(full)；
wait(mutex)；
nextc：= buffer[out]；
out:=(out +1) % n；
signal(mutex)；
signal(empty)；
consume the item in nextc；
}while(TRUE);
 }
```
